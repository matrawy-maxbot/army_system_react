### دالة الحفظ
```javascript
const handleSave = async () => {
    // التحقق من كلمة المرور
    if (showPasswordSection && !handlePasswordSubmit()) return;
    
    // التحقق من النسخ الاحتياطية
    if (maxBackupsError || maxBackups < 1 || maxBackups > 50) return;
    
    // حفظ الإعدادات
    const settings = {
        // ... الإعدادات الحالية
        hasPassword: showPasswordSection ? true : hasPassword,
        password: showPasswordSection ? newPassword : undefined,
        autoStart: autoStart,
        maxBackups: parseInt(maxBackups)
    };
    
    await settingsAPI.saveSettings(settings);
}
```

#### 2. التحقق من النسخ الاحتياطية
```javascript
const handleMaxBackupsChange = (value) => {
    const num = parseInt(value);
    
    if (isNaN(num)) setMaxBackupsError('الرجاء إدخال رقم صحيح');
    else if (num < 1) setMaxBackupsError('الحد الأدنى هو 1');
    else if (num > 50) setMaxBackupsError('الحد الأقصى هو 50');
    else setMaxBackupsError('');
    
    setMaxBackups(value);
}
```
